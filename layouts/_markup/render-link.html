{{- /* 本文に URL だけを書いた行は、相手ページの OGP（タイトル・説明・サムネ）を取得してリンクカードにする */ -}} {{- $u := .Destination -}} {{- $bare := and (or (hasPrefix $u "http://") (hasPrefix $u "https://")) (eq (.Text | string | plainify | htmlUnescape) $u) -}} {{- if and $bare (findRE `(youtube\.com/(watch|shorts/)|youtu\.be/|vimeo\.com/)` $u 1) -}} {{- partial "video-embed.html" $u -}} {{- else if $bare -}} {{- $host := (urls.Parse $u).Host -}} {{- $og := dict -}} {{- with try (resources.GetRemote $u (dict "headers" (dict "User-Agent" "Mozilla/5.0 (compatible; LinkCardBot; +https://yumagarments.github.io/)"))) -}} {{- with .Err -}}{{- warnf "linkcard: %s を取得できませんでした" $u -}} {{- else with .Value -}} {{- $html := .Content -}} {{- range slice "og:title" "og:description" "og:image" "og:site_name" "twitter:image" "description" -}} {{- $key := . -}} {{- with findRE (printf `]*(?:property|name)=["']%s["'][^>]*>` $key) $html 1 -}} {{- with findRESubmatch `content=["']([^"']*)["']` (index . 0) 1 -}} {{- $og = merge $og (dict $key (index (index . 0) 1 | htmlUnescape)) -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $og.og_title -}} {{- with findRESubmatch `(?s)]*>(.*?)` $html 1 -}}{{- $og = merge $og (dict "title" (index (index . 0) 1 | htmlUnescape | strings.TrimSpace)) -}}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $title := or (index $og "og:title") (index $og "title") $host -}} {{- $desc := or (index $og "og:description") (index $og "description") "" -}} {{- $site := or (index $og "og:site_name") $host -}} {{- $img := or (index $og "og:image") (index $og "twitter:image") "" -}} {{- if and $img (hasPrefix $img "//") -}}{{- $img = printf "https:%s" $img -}}{{- end -}} {{- if and $img (hasPrefix $img "/") -}}{{- $p := urls.Parse $u -}}{{- $img = printf "%s://%s%s" $p.Scheme $p.Host $img -}}{{- end -}} {{- $thumb := "" -}} {{- with $img -}} {{- $thumb = . -}} {{- /* 取得できれば正方形に切り抜いて自サイトに保存（相手サイトの画像が消えても表示される） */ -}} {{- with try (resources.GetRemote .) -}}{{- with .Value -}}{{- if in (slice "image/jpeg" "image/png" "image/webp" "image/gif") .MediaType.Type -}} {{- $thumb = (.Fill "320x320 Center webp").RelPermalink -}} {{- end -}}{{- end -}}{{- end -}} {{- end -}} {{ $title }}{{ with $desc }}{{ . | truncate 100 }}{{ end }}{{ $site }}{{ with $thumb }}{{ end }} {{- else -}} {{ .Text | safeHTML }} {{- end -}}
